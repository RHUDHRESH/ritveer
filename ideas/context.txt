### 2.1 SQLite Schema Design ✅
**File:** `src/karigar/schemas/models.py`

```python
from sqlalchemy import Column, String, Integer, Float, Text, DateTime, JSON, ForeignKey
from sqlalchemy.ext.declarative import declarative_base
from datetime import datetime

Base = declarative_base()

class Artisan(Base):
    __tablename__ = "artisans"
    id = Column(String, primary_key=True)
    name = Column(String)
    phone = Column(String)
    location = Column(String)
    created_at = Column(DateTime, default=datetime.utcnow)

class MaterialRequest(Base):
    __tablename__ = "material_requests"
    id = Column(String, primary_key=True)
    artisan_id = Column(String, ForeignKey("artisans.id"))
    material = Column(String)
    quantity = Column(Float)
    budget = Column(Float)
    timeline = Column(String)
    status = Column(String)  # pending, quoted, ordered, delivered
    created_at = Column(DateTime, default=datetime.utcnow)

class SupplierQuote(Base):
    __tablename__ = "supplier_quotes"
    id = Column(String, primary_key=True)
    request_id = Column(String, ForeignKey("material_requests.id"))
    supplier_name = Column(String)
    price = Column(Float)
    delivery_days = Column(Integer)
    rating = Column(Float)
    selected = Column(Integer, default=0)

class Order(Base):
    __tablename__ = "orders"
    id = Column(String, primary_key=True)
    request_id = Column(String, ForeignKey("material_requests.id"))
    quote_id = Column(String, ForeignKey("supplier_quotes.id"))
    upi_link = Column(String)
    po_path = Column(String)
    status = Column(String)
    created_at = Column(DateTime, default=datetime.utcnow)

class MicroStore(Base):
    __tablename__ = "micro_stores"
    id = Column(String, primary_key=True)
    artisan_id = Column(String, ForeignKey("artisans.id"))
    product_name = Column(String)
    price = Column(Float)
    image_path = Column(String)
    qr_code_path = Column(String)
    store_url = Column(String)
    created_at = Column(DateTime, default=datetime.utcnow)

class Ledger(Base):
    __tablename__ = "ledger"
    id = Column(String, primary_key=True)
    artisan_id = Column(String, ForeignKey("artisans.id"))
    order_id = Column(String, ForeignKey("orders.id"))
    type = Column(String)  # debit, credit
    amount = Column(Float)
    description = Column(Text)
    timestamp = Column(DateTime, default=datetime.utcnow)
```

### 2.2 Vector Memory (ChromaDB)
**File:** `src/karigar/memory/vector_store.py`

```python
import chromadb
from chromadb.config import Settings

class VectorMemory:
    def __init__(self, persist_directory="./chroma_data"):
        self.client = chromadb.Client(Settings(
            persist_directory=persist_directory,
            anonymized_telemetry=False
        ))
        self.suppliers = self.client.get_or_create_collection("suppliers")
        self.products = self.client.get_or_create_collection("products")
    
    def store_supplier(self, supplier_id, name, materials, pricing, rating):
        doc = f"{name} supplies {materials} at {pricing}. Rating: {rating}"
        self.suppliers.add(
            documents=[doc],
            ids=[supplier_id],
            metadatas=[{"name": name, "rating": rating}]
        )
    
    def search_suppliers(self, query, n_results=5):
        results = self.suppliers.query(
            query_texts=[query],
            n_results=n_results
        )
        return results
```

### 3.1 Supplier Search Tool (Perplexity Fallback)
**File:** `src/karigar/tools/supplier_search.py`

```python
import httpx
from typing import Optional

class SupplierSearchTool:
    def __init__(self, perplexity_api_key: Optional[str] = None):
        self.api_key = perplexity_api_key
        self.local_suppliers = self._load_local_suppliers()
    
    def _load_local_suppliers(self):
        # Local supplier database (CSV/JSON)
        return {
            "cement": [
                {"name": "BuildMart Delhi", "price": 350, "delivery": 2, "rating": 4.5},
                {"name": "QuickSupply", "price": 380, "delivery": 1, "rating": 4.2}
            ],
            "bricks": [
                {"name": "Brick Co", "price": 8, "delivery": 3, "rating": 4.7}
            ]
        }
    
    async def search(self, material: str, location: str) -> list:
        # Try local first
        results = self.local_suppliers.get(material.lower(), [])
        
        # Fallback to Perplexity if enabled and local results insufficient
        if len(results) < 2 and self.api_key:
            results.extend(await self._perplexity_search(material, location))
        
        return results
    
    async def _perplexity_search(self, material: str, location: str):
        async with httpx.AsyncClient() as client:
            response = await client.post(
                "https://api.perplexity.ai/chat/completions",
                headers={"Authorization": f"Bearer {self.api_key}"},
                json={
                    "model": "llama-3.1-sonar-small-128k-online",
                    "messages": [
                        {
                            "role": "user",
                            "content": f"Find 3 suppliers for {material} in {location} with prices"
                        }
                    ]
                }
            )
            # Parse and structure response
            return self._parse_perplexity_response(response.json())
```

### 3.2 UPI Link Generator
**File:** `src/karigar/tools/payment_tools.py`

```python
from urllib.parse import quote

class PaymentTools:
    @staticmethod
    def generate_upi_link(amount: float, upi_id: str, name: str, note: str):
        base_url = "upi://pay"
        params = f"?pa={upi_id}&pn={quote(name)}&am={amount}&tn={quote(note)}"
        return base_url + params
    
    @staticmethod
    def generate_qr_code(upi_link: str, save_path: str):
        import qrcode
        qr = qrcode.QRCode(version=1, box_size=10, border=4)
        qr.add_data(upi_link)
        qr.make(fit=True)
        img = qr.make_image(fill_color="black", back_color="white")
        img.save(save_path)
        return save_path
```

```python
from reportlab.lib.pagesizes import letter
from reportlab.pdfgen import canvas

class PDFGenerator:
    @staticmethod
    def generate_po(order_data: dict, save_path: str):
        c = canvas.Canvas(save_path, pagesize=letter)
        c.setFont("Helvetica-Bold", 16)
        c.drawString(100, 750, "PURCHASE ORDER")
        
        c.setFont("Helvetica", 12)
        y_position = 700
        for key, value in order_data.items():
            c.drawString(100, y_position, f"{key}: {value}")
            y_position -= 20
        
        c.save()
        return save_path
```

```python
from karigar.schemas.state import KarigarState
from langchain_openai import ChatOpenAI

class IntakeAgent:
    def __init__(self, llm: ChatOpenAI):
        self.llm = llm
    
    def process(self, state: KarigarState) -> dict:
        user_input = state["messages"][-1].content
        
        prompt = f"""Extract structured data from:
        {user_input}
        
        Return JSON with: material, quantity, budget, timeline, location"""
        
        response = self.llm.invoke(prompt)
        # Parse response into structured dict
        material_request = self._parse_response(response.content)
        
        return {"material_request": material_request, "status": "intake_complete"}
```

### 4.3 Orchestrator (Wire All Agents)
**File:** `src/karigar/orchestrators/workflow.py`

```python
from langgraph.graph import StateGraph, START, END
from karigar.schemas.state import KarigarState
from karigar.agents.intake_agent import IntakeAgent
# Import other agents...

class KarigarWorkflow:
    def __init__(self):
        self.graph = StateGraph(KarigarState)
        self._build_graph()
    
    def _build_graph(self):
        # Add nodes
        self.graph.add_node("intake", IntakeAgent().process)
        self.graph.add_node("supplier", SupplierAgent().process)
        self.graph.add_node("commit", CommitAgent().process)
        self.graph.add_node("sales", SalesAgent().process)
        self.graph.add_node("cash", CashAgent().process)
        
        # Add edges
        self.graph.add_edge(START, "intake")
        self.graph.add_edge("intake", "supplier")
        self.graph.add_edge("supplier", "commit")
        self.graph.add_edge("commit", "sales")
        self.graph.add_edge("sales", "cash")
        self.graph.add_edge("cash", END)
        
        self.compiled = self.graph.compile()
    
    async def run(self, initial_state: dict):
        return await self.compiled.ainvoke(initial_state)
```

### 6.3 Request Form
**File:** `src/components/RequestForm.jsx`

```jsx
import React, { useState } from 'react';
import axios from 'axios';

export default function RequestForm() {
  const [formData, setFormData] = useState({
    artisan_id: '',
    message: ''
  });
  const [result, setResult] = useState(null);

  const handleSubmit = async (e) => {
    e.preventDefault();
    try {
      const response = await axios.post('http://localhost:8000/api/request', formData);
      setResult(response.data);
    } catch (error) {
      console.error(error);
    }
  };

  return (
    <div className="max-w-2xl mx-auto p-6 bg-white rounded-lg shadow-lg">
      <h2 className="text-2xl font-bold mb-4">Request Materials</h2>
      <form onSubmit={handleSubmit} className="space-y-4">
        <input
          type="text"
          placeholder="Your Artisan ID"
          className="w-full p-3 border rounded"
          value={formData.artisan_id}
          onChange={(e) => setFormData({...formData, artisan_id: e.target.value})}
        />
        <textarea
          placeholder="I need 100 kg cement in Delhi by next week, budget 5000 rupees"
          className="w-full p-3 border rounded h-32"
          value={formData.message}
          onChange={(e) => setFormData({...formData, message: e.target.value})}
        />
        <button
          type="submit"
          className="w-full bg-blue-500 text-white p-3 rounded hover:bg-blue-600"
        >
          Submit Request
        </button>
      </form>
      
      {result && (
        <div className="mt-6 p-4 bg-green-50 rounded">
          <h3 className="font-bold">Success!</h3>
          <pre>{JSON.stringify(result, null, 2)}</pre>
        </div>
      )}
    </div>
  );
}
```
